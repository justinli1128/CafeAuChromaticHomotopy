\begin{definition}
    A finite $p$-local spectrum $X$ has \textbf{type $n$} if $K(n-1)_*X\cong 0$ and $K(n)_*X\not \cong 0$.

    For a type $n$ spectrum $X$, a \textbf{$v_n$ self map} is a map $v: \Sigma^{d}X\to X$ for some $d\in \Z$ such that 
    \begin{align*}
        K(n)_*v: K(n)_{*-d}X\to K(n)_* X
    \end{align*}is an isomorphism, and 
    \begin{align*}
        K(m)_*v: K(m)_{*-d}X\to K(m)_* X
    \end{align*}is nilpotent for $m\neq n$.
\end{definition}

    are thick subcategory of the category of finite $p$-local spectrum.
