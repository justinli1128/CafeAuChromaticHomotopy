\begin{definition}
\cite{PreRudyak}
    A \textbf{reduced generalized cohomology} is a contravariant functor $\HH^*: \Spc_*^{op}\to \mathrm{grAb}$ from the homotopy category of spaces to the category of $\Z$-graded abelian groups, such that 
    \begin{itemize}
        \item (Additive) For every set of pointed spaces $X_\alpha$ for $\alpha \in A$, the canonical homomorphism $\HH^*(\bigvee_{\alpha \in A}X_\alpha)\to \prod_{\alpha \in A} \HH^*(X_\alpha)$ is an isomorphism.
        \item (Mayer-Vietoris) For every homotopy cofiber sequence in pointed space $X\to Y\to Z$, there is long exact sequence
        \begin{align*}
        ...\to \HH^{n-1}(X)\xrightarrow{\delta_{n-1}} \HH^n(Z)\to \HH^n(Y) 
        \\ \to \HH^n(X)\xrightarrow{\delta_n} \HH^{n+1}(Z)\to ...
        \end{align*}
    \end{itemize}

    A \textbf{reduced generalized homology} is a covariant functor $\HH_*: \Spc_*\to \mathrm{grAb}$ from the homotopy category of spaces to the category of $\Z$-graded abelian groups, such that 
     \begin{itemize}
        \item (Additive) For every set of pointed spaces $X_\alpha$ for $\alpha \in A$, the canonical homomorphism $\coprod_{\alpha \in A} \HH_*(X_\alpha)\to \HH_*(\bigvee_{\alpha \in A}X_\alpha)$ is an isomorphism.
        \item (Mayer-Vietoris) For every homotopy cofiber sequence in pointed space $X\to Y\to Z$, there is long exact sequence
        \begin{align*}
        ...\to \HH^{n+1}(Z)\xrightarrow{\delta_{n}} \HH_n(X)\to \HH_n(Y) 
        \\ \to \HH_n(Z)\xrightarrow{\delta_{n-1}} \HH_{n-1}(X)\to ...
        \end{align*}
    \end{itemize}

\end{definition}

A morphism of cohomology (resp. homology) are just natural transformations.

\begin{remark}
    Given a spectrum $E$, the functor $E^n(X):=[\Sigma^{-n}\Sigma^\infty X, E]$ (where $\Sigma^{-n}=\Omega^n$ for $n>0$), is a cohomology. 

    The additive property follows from 
    \[
    [\Sigma^{-n}\Sigma^\infty (\bigvee_{\alpha\in A}X_\alpha), E]\cong [\bigvee_{\alpha\in A}\Sigma^{-n}\Sigma^\infty (X_\alpha), E]\cong \prod_{\alpha\in A}[\Sigma^{-n}\Sigma^\infty (X_\alpha), E]]
    \]

    The Mayer-Vietoris property follows from that a homotopy cofiber sequence is taken to homotopy cofiber sequence 
    \begin{align*}
    ...\to \Sigma^{-(n+1)}\Sigma^\infty Z\to \Sigma^{-n}\Sigma^\infty X\to  \Sigma^{-n}\Sigma^\infty Y
    \\ \to  \Sigma^{-n}\Sigma^\infty Z\to  \Sigma^{-(n-1)}\Sigma^\infty X\to ... 
    \end{align*}
    Then there is long exact sequence
    \begin{align*}
    ...\to E^{n-1}(X)=[\Sigma^{-(n-1)}\Sigma^\infty X, E]\to E^n(Z)=[\Sigma^{-n}\Sigma^\infty Z, E]
    \\ \to  E^n(Y)\to  E^n(Z)\to  E^{n-1}(X)\to ... 
    \end{align*}

    Equivalently, the functor $E_n(X):=[\Sigma^n, \Sigma^\infty X \wedge E]$ is a generalized homology.
\end{remark}

\begin{theorem}(Adams Version of Brown's Representability Theorem)
\cite{PreSwitzer}

Every cohomology $\HH^*$ (resp. homology $\HH_*$) there is a spectra $E$ and a natural isomorphism $\alpha: E^*\to \HH^*$ (resp. $\alpha: E_*\to \HH_*$). Moreover, given another spectrum $E'$ and a natural transformation $\eta: E'^*(-)\to \HH^*(-)$ (resp. $\eta: E'_*(-)\to \HH_*(-)$), there exists a spectra map $\eta': E'\to E$ such that $\alpha\circ \eta'=\eta$.
\end{theorem}

\begin{remark}
Warning: The natural transformations between cohomologies and homology do not lift to spectra maps uniquely. There exists morphisms that is not nullhomotopic between spectra that determine the zero morphism between the homologies, such map is called a \textbf{phantom map} \cite{PreRudyak}. However, under certain conditions \cite{ChrLurie}, the lifting exists uniquely. More on this later.
\end{remark}

\begin{example}
Let $A$ be an abelian group, and let $\tilde H^*(X;A)$ the ordinary reduced cohomology with coefficient in $A$. Let $HA$ be the spectra representing $\tilde H^*(X;A)$, then we have the following property
\begin{align*}
[\Sigma^{-n} \Sigma ^\infty S^m , HA]=[S^m, \Omega^\infty \Omega^{-n} HA]
\\=[S^m, HA_n]
\\= \tilde H^n(S^m;A)
\\= \begin{cases}
0, n\neq m
\\ A, n=m
\end{cases}
\end{align*}In other words, $HA_n$ is an eilenberg-maclane space of type $(A, n)$. Notice, $\Omega K(A, n+1)\simeq K(A,n)$, so this indeed determines a $\Omega$-spectrum, which we refer to as the \textbf{Eilenberg-Maclane spectrum of} $A$.
\end{example}

Given spectra $E$ and $F$, the multiplication of integers on $[E,F]$ is determined by 
\begin{align*}
\Z\times [E, F]\cong [S, S]\times [E, F]\xrightarrow{(n, f)\mapsto (n\wedge f)} [S\wedge E,S\wedge F]=[E, F]
\end{align*}

\begin{remark}
    From now on, for every spectra $E$, we denote $E_*:=E_*(S^0)=[S^*, E]$ and refer to it as the \textbf{homological coefficient group} of $E$. Notice, we can define $E^*:=E^*(S^0)$ and refer to it as the \textbf{cohomological coefficient group} of $E$, and there is isomorphism of graded abelian $E_*=E^{-*}$.

    If $E$ is a commutative ring spectra, $E^*$ and $E_*$ are both graded rings, with multiplication determined by 
    \begin{align*}
        E_{-n}\otimes E_{-m}=E^n\otimes E^m=[S^{-n}, E]\otimes [S^{-m}, E]\xrightarrow{(r, r')\mapsto (r\wedge r')}[S^{-(n+m)}, E\wedge E]
        \\ \xrightarrow{\psi\circ -} [S^{-(n+m)}, E]=E^{n+m}=E_{(-n)+(-m)}
    \end{align*}

    Moreover, $E^*(-)$ (resp. $E_*(-)$) values in graded left $E^*$-module (resp. left $E_*$-module) determined pointwise on $X$ by
    \begin{align*}
       E_{-n}\otimes E_{-m}(X)= E^n\otimes E^m(X)=[S^{-n}, E]\otimes [\Sigma^{-m}X, E]
       \\ \xrightarrow{(r, m)\mapsto (r\wedge m)}[\Sigma^{-(n+m)}X, E]=E^{n+m}(X)=E_{(-n)+(-m)}(X)
    \end{align*}(similarly for right $E_*$-module structure)
\end{remark}



\begin{definition}
For a commutative ring spectrum $E$, the cohomology $E^*(-)$ with domain in CW complexes values in graded commutative ring, with the multiplication pointwise at each CW complex $X$ determined by 
    \begin{align*}
E^n(X)\otimes E^m(X)=[\Sigma^{-n}\Sigma^∞ X, E]\otimes[\Sigma^{-m}\Sigma^∞ X, E]\to [\Sigma^{-(n+m)}\Sigma^∞ (X\wedge X), E\wedge E]
\\ \xrightarrow{-\circ \nabla} [\Sigma^{-(n+m)}\Sigma^∞ X, E\wedge E]\xrightarrow{\psi \circ -}[\Sigma^{-(n+m)}\Sigma^∞ X, E]=E^{n+m}(X)
    \end{align*}where $\psi$ is the multiplication and $\nabla: X\to X\wedge X$ is the diagonal map.
\end{definition}

\begin{proposition}
    The ring structure of $E^*(-)$ determines a graded $E^*$-algebra structure.
\end{proposition}

If $R$ is a commutative ring, then $H^*(-;R)$ takes value in graded commutative ring, with multiplication determined by the cup product of $R$-coefficient singular cochains. 
\begin{proposition}
\cite{PreRudyak} 
There is commutative ring spectra structure on $HR$, such that the multiplication induces the cup product multiplication on $H^*(-;R)$.
\end{proposition}

\begin{theorem}((cohomological) Atiyah-Hirzebruch Spectral Sequence)\cite{PreMcCleary}
    For a commutative ring spectrum $E$, and a CW complex $X$, there is a conditionally convergent multiplicative spectral sequence
    \begin{align*}
        E_2^{p,q}=H^p(X; E^q)\implies E^{p+q}(X)
    \end{align*}with differential $d_r^{p,q}: E_r^{p,q}\to E_r^{p+r,q-r+1}$, that is functorial in $X$. Moreover, this is strongly convergent for finite CW $X$.
\end{theorem}
