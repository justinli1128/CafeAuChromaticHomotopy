\begin{definition}
  Given two cohomology $E$ and $F$, a \textbf{stable cohomology operation of degree $k$} is a natural transformation $\varphi: E^*(-) \to F^{*+k}(-)$ such that 
  \begin{center}
\begin{tikzcd}
E^{*+1}(\Sigma X) \arrow[r, "\varphi"] \arrow[d, "\delta_E"] & F^{*+k+1}(\Sigma X) \arrow[d, "\delta_F"] \\
E^*(X) \arrow[r, "\varphi"]                                & F^{*+k}(X)                             
\end{tikzcd}
  \end{center}commutes, where $\delta_E: E^{*+1}(\Sigma X)\to E^{*}(X)$ is the connecting isomorphism of the Mayer-Vietoris sequence for homotopy cofiber sequence $X\to *\to \Sigma X$.
\end{definition}
