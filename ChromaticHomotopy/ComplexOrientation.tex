\begin{definition}
    A \textbf{complex orientable} spectrum is a multiplicative spectrum $E$ such that $\tilde E^2(\mathbb{C}P^\infty)\to \tilde E^2(\mathbb{C}P^1)\cong E^0$ is surjective.

    Since $E$ is multiplicative, $E^0$ has a generator $1_E$, which correspond to a generator $t\in E^2(\mathbb{C}P^1)$. A \textbf{complex orientation} is a lift of $t$ in $E^2(\mathbb{C}P^\infty)$. A \textbf{complex oriented cohomology} is a tuple $(E,t_E)$ where $E$ is a complex orientable cohomology with a complex orientation
\end{definition}

\begin{example}
    Let $E=HR$ for some commutative ring $R$, then $HR$ is complex oriented with $t\in H^2(S^2; R)$ with $H^*(\mathbb{C}P^∞;R)\cong R[[t]]$.
\end{example}
\begin{theorem}
    \cite{PreRavenelGreen}\cite{PreAdams} \cite{ChrHopkins}
    Let $(E, t)$ be a complex oriented cohomology, then there exists isomorphism
    \begin{align*}
        E^*(\mathbb{C}P^∞)\cong E^*[[t]]
    \end{align*}and $  E^*(\mathbb{C}P^n)\cong E^*[t]/(t^{n+1})$
\end{theorem}
See \cite{ChrLurie} for a more general statement not using Aityah-Hirzebruch spectral sequence.
\begin{proof}
    We use the Atiyah-Hirzebruch spectral sequence for $\mathbb{C}P^n$, with $E_2$-page $H^p(\mathbb{C}P^n; E^q)$, which is strongly convergent as $\mathbb{C}P^n$ is finite.  

    First of all, $H^p(\mathbb{C}P^n; \Z)$ are free, therefore $H^p(\mathbb{C}P^n; A)=H^p(\mathbb{C}P^n; \Z) \otimes A$. Since $H^*(\mathbb{C}P^n; \Z)=\Z[t]/(t^{n+1})$ for $|t|=2$. Therefore, the second page of the AHSS is
  
\[
\begin{array}{c|cccccc}
q & \\
\vdots & \vdots & & \vdots & & & \vdots \\
2 & E^{2} & * & E^{2}t & * & \cdots & E^{2}t^n \\
1 & E^{1} & * & E^{1}t & * & \cdots & E^{1}t^n \\
0 & E^{0} & * & E^{0}t & * & \cdots & E^{0}t^n \\
\hline
p & 0 & 1 & 2 & 3 & \cdots & 2n
\end{array}
\]
    The spectral sequence is multiplicative and the differentials are $E^*$. Every element in the $E_2$-page is in the form $et^k$ or $e$ for $e\in E^*$, and $d_2(e)=ed_2(1_E)$ and $d_2(et^k)=ket^{k-1}d_2t$. Therefore, $d_2$ is determined entirely by $d_2(1_E)\in E^{-1}$ and $d_2t\in E^{-1} t^2$.

    Recall that AHSS is functorial, and AHSS for $\mathbb{C}P^n$ can be constructed from filtration determined by the CW structure $\mathbb{C}P^k\subseteq \mathbb{C}P^{k+1}\subseteq \mathbb{C}P^n$. We will use induction to show that the spectral sequence for $\mathbb{C}P^n$ collapses on $E_2$-page, that is $d_r=0$ for all $r$. 
    
    The assumption for $\mathbb{C}P^1\simeq S^2$ comes from the collapsing map $S^2\to *$. The inclusion and collapsing $*\to S^2\to *$ is obviously the identity and induces isomorphism. Therefore, the trivialness of $d_2$ on $E_2(*)$ implies that of $E_2(S^2)$, because there are no other differentials that are not necessarily trivial besides the one for $p=0$. For $E_n$ page of $S^2$, no differentials has both nontrivial target and domain, so the sequence collapses on $E_2$.

    Suppose $\mathbb{C}P^{n-1}$ has collapsing $E_2$ page. The inclusion map $\mathbb{C}P^{n-1}\to \mathbb{C}P^n$, which by assumption takes $t\in E^2(\mathbb{C}P^{n})$ to $t\in E^2(\mathbb{C}P^{n-1})$ determines an isomorphism on $E^{p,q}_2$ for $p<2n$. Therefore, the only possible nontrivial $d_2$ differentials are the ones 
    \[
    d_2:H^{2n-2}(\mathbb{C}P^n, E^q)\to H^{2n}(\mathbb{C}P^n, E^{q-1})
    \]
    The collapsing map $\mathbb{C}P^n\to \mathbb{C}P^n/\mathbb{C}P^{n-1}\simeq S^{2n}$ induces $\varphi: E_r(S^{2n})\to E_r^{\mathbb{C}P^n}$. Because it lives on the cofiber sequence 
    \[
    \mathbb{C}P^{n-1}\to \mathbb{C}P^n\to S^{2n}
    \]
    And $E_r^{2n,q}(\mathbb{C}P^{n-1})=0$. The map $\varphi$ is an isomorphism for all $r$ on $p=2n$. Suppose $d_2(t^{n-1})=et^n$ for $e\neq 0$, then on the $E_3$ page, we have the quotient map $E^*\to E^*/e$ is an isomorphism, which is absurd.

    No more nontrivial differentials exists after $E_2$ so it collapses.

    To show $E^*(\mathbb{C}P^n)\cong E^*[t]/(t^{n+1})$, we once again look at the cofiber sequence
    \[
    \mathbb{C}P^{n-1}\to \mathbb{C}P^n\to S^{2n}
    \]
    which induces 
    \[
        ...\to E^{*-1} \mathbb{C}P^{n-1}\to E^{*} S^{2n}\to E^*\mathbb{C}P^{n}\to E^{*}\mathbb{C}P^{n-1} \to E^{*+1} S^{2n}\to...
    \]
    If we assume $E^*(\mathbb{C}P^{n-1})\cong E^*[t]/(t^{n})$, then everything but $E^*\mathbb{C}P^{n}$ in this long exact sequence are $E^*$-free, so they splits, and we have $t^{n}$ is a generator in $E^*\mathbb{C}P^{n}$, this statement then follows.

    Taking one step further, from Milnor's sequence
\begin{align*}
    0\to {\lim}^1E^{*-1}(\mathbb{C}P^n)\to E^*(\mathbb{C}P^\infty)\to {\lim}E^*(\mathbb{C}P^n)\to 0
\end{align*} and the fact $E^{*}(\mathbb{C}P^{n+1})\to E^{*}(\mathbb{C}P^n)$ is surjective, we have 
\begin{align*}
    E^*(\mathbb{C}P^\infty)\cong E^*[[t]]
\end{align*}
\end{proof}

The name complex orientation has a geometric meaning. We know complex line bundle is classified by the space $\mathbb{C}P^\infty\simeq BU(1)$. Given a complex oriented cohomology, $t$ is then a characteristic class of universal complex line bundle.

\begin{lemma}
    $E^*(\mathbb{C}P^∞\times \mathbb{C}P^∞)\cong E^*[[x,y]]$
\end{lemma}
\begin{proof}
    This is because $E^*(\mathbb{C}P^∞)$ is $E^*$-free.
\end{proof}
The tensor product of line bundle $\psi:\mathbb{C}P^∞\times \mathbb{C}P^∞\to \mathbb{C}P^∞$ then induces a map 
\begin{align*}
    \psi_E:E^*(\mathbb{C}P^∞)\cong E^*[[t]]\to  E^*(\mathbb{C}P^∞\times \mathbb{C}P^∞)\cong E^*[[x,y]]
\end{align*}
The generator $t$ then correspond to a formal power series in $x,y$, $F_E(x,y):=\psi_E(t)$.

\begin{definition}
    A \textbf{formal group law} over a ring $R$ is a formal power series $F(x,y)$ such that 
    \begin{itemize}
        \item $F(x,0)=F(0,x)=x$
        \item $F(x,y)=F(y,x)$
        \item $F(x,F(y,z))=F(F(x,y),z)$
    \end{itemize}
For $R$ graded, we require $F(x,y)$ to be homogeneous of degree $2$.
\end{definition}
We will write $x+_Fy:=F(x,y)$.

\begin{lemma}
    \cite{PreRudyak} Let $E$ be a complex oriented cohomology, $F_E(x,y)$ is a graded formal group law.
\end{lemma}
\begin{proof}
   This follows from the properties of tensor product of line bundle being unital, associative and commutative. 
\end{proof}

\begin{example}
    Let $E=HR$ for some ring $R$, $f_E(x,y)=x+y$, as it is the only possible graded formal group law.
   \\ Let $E=KU$ be the complex $K$-theory, we choose $t\in KU^2(\mathbb{C}P^\infty)$ to be the one that correspond to $\beta(\xi_1-1)$, where $\xi_1$ is the universal line bundle, $1$ is the trivial rank $1$ bundle over $\mathbb{C}P^\infty$, and $\beta$ is the Bott map $\beta: KU^*\to KU^{*+2}$. The pullback of line bundle $\mathcal{L}$ on $\mathbb{C}P^\infty$ pullback to $p_1^*\mathcal{L}\otimes p_2^*\mathcal{L}$ on $\mathbb{C}P^\infty\times \mathbb{C}P^\infty$.
   
   We have that given line bundle
    \begin{align*}
\psi^*(\beta(\xi_1-1))= \beta(\psi^*\xi_1-1)
\\= \beta(p_1^*\xi_1\otimes p_2^*\xi_1-1)
\\=\beta(p_1^*\xi_1-1)\otimes  (p_2^*\xi_1-1)+ \beta[(p_1^*\xi_1-1)\otimes 1 ]+  \beta[1\otimes (p_2^*\xi_1-1)]
    \end{align*} So $F_K(x,y)=x+y+\beta^{-1}xy $
\end{example}

There exists map
\[
    BU(1)^{\times n}\to BU(n)
\]determined by the map that classifies the direct sum of $n$ complex line bundles. Since permuting direct summand does not change the vector bundles, this is a $\Sigma_n$-invariant map.

\begin{proposition}
    \cite{PreSwitzer}\cite{ChrLurie}
Given a complex oriented cohomology $E$, we have the map $  BU(1)^{\times n}\to BU(n)$ determines an isomorphism 
\[
E^*(BU(n))\cong E^*(BU(1)^{\times n})^{\Sigma_n}\cong E^*[[t_1, t_2,...,t_n]]^{\Sigma_n}\cong E^*[[c_1,...,c_n]]
\]
and 
\[
E^*(BU)\cong E^*[[c_1,c_2,...]]
\]for $|c_i|=2i$
\end{proposition}

These $c_i$ are the $i$th Chern class of complex vector bundle.

\begin{proposition}
    Given a complex oriented cohomology $E$, we have 
    \begin{align*}
        E_*(BU(n))=\mathrm{Sym}_{\leq n}E_*\{b_1,...\}
    \end{align*}the degree less or equal $n$ part of the symmetric algebra with generator $b_i$, and
    \begin{align*}
        E_*(BU)= E_*[b_1,...]
    \end{align*} 
    as $E_*$-module.
\end{proposition}
\begin{proof}
This follows from the fact that everything are free.
\end{proof}
There is indeed an algebra structure on $E_*BU$ which justifies this notation (more on this later)
