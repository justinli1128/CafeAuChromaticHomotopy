Recall the definition of a formal group law.

\begin{definition}
    A \textbf{formal group law} over a ring $R$ is a formal power series $F(x,y)$ such that 
    \begin{itemize}
        \item $F(x,0)=F(0,x)=x$
        \item $F(x,y)=F(y,x)$
        \item $F(x,F(y,z))=F(F(x,y),z)$
    \end{itemize}
For $R$ graded, we require $F(x,y)$ to be homogeneous of degree $2$.
\end{definition}
\begin{remark}
    Given an ungraded formal group law $F$ over $R$ determines a continuous ring map $R[[t]]\to R[[x,y]]$, because, $F(x,y)$ has no constant term, so $F(x,y)^n\to 0$. Therefore, there exists a map
     $\mathrm{Spf}(R[[x]])\times \mathrm{Spf}(R[[y]])\to \mathrm{Spf}(R[[t]])$, which is by the above definition, is a group object in the category of affine formal schemes. More on this later.
\end{remark}

\begin{definition}
    Let $F$ be a formal group law over $R$ and $G$ be a formal group law over $S$, a \textbf{morphism of formal group law} $\varphi: F/R\to G/S$ contains a ring map $\psi: R\to S$, and a formal power series $f(t)\in S[[t]]$ such that 
    \begin{align*}
        f(\psi^*F(x,y))=G(f(x),f(y))
    \end{align*}
    where $\psi^*F$ is the formal group law over $S$ by taking the coefficient of $F$ to $S$ along $\psi$.  

    An \textbf{isomorphism of formal group law} is a morphism of formal group law with an inverse, and a \textbf{strict isomorphism of formal group law} is an isomorphism $\varphi=(\psi, f): F/R\to G/S$, such that $f'(0)=1$.
\end{definition}

Given formal group law $F$, we denote $x+_F y:=F(x,y)$.


\begin{definition}
    Let $F$ be a formal group law over $R$ and $G$ be a formal group law over $S$, a \textbf{morphism of formal group law} $\varphi: F/R\to G/S$ contains a ring map $\psi: R\to S$, and a formal power series $f(t)\in S[[t]]$ such that 
    \begin{align*}
        f(\psi^*F(x,y))=G(f(x),f(y))
    \end{align*}
    where $\psi^*F$ is the formal group law over $S$ by taking the coefficient of $F$ to $S$ along $\psi$.  

    An \textbf{isomorphism of formal group law} is a morphism of formal group law with an inverse, and a \textbf{strict isomorphism of formal group law} is an isomorphism $\varphi=(\psi, f): F/R\to G/S$, such that $f'(0)=1$.
\end{definition}


\begin{definition}
    The functor $\mathbf{fgl}: \Ring\to \Sets$ is such that 
    \[
    \mathbf{fgl}(R)=\{F\in R[[x,y]]: F \text{ is a formal group law over $R$}\}
    \]and for $\psi: R\to S$, $\mathbf{fgl}(\psi):\mathbf{fgl}(R)\to \mathbf{fgl}(S)$ takes $F$ to $\psi^*F$.

\end{definition}
Of course, every definitions here can be generalized for graded ring and graded forma group law by adding the condition that every formal power series are homogeneous of the same degree.
\begin{proposition}
\cite{ChrHopkins}
The functor $\mathbf{fgl}$ is corepresentable by a ring $L$, that is $\fgl=\Hom_{\Ring}(L,-)$.
\end{proposition}
\begin{proof}
    First of all, $\mathbf{fgl}$ is a subfunctor of the formal power series functor $\mathbf{pf}$ that takes a ring $R$ to the set of two variable formal power series over $R$. We have that $\mathbf{pf}$ is corepresented by ring $\Z[c_{00},c_{01},c_{10}, c_{11}...]$ with the universal power series
    \[
    F(x,y)=\sum_{i,j\in \mathbb{N}^2}c_{ij}x^iy^j
    \] Now, $F$ is a formal group law if and only if its coefficients satisfy the defining relations. Denote the ideal of the relations as $I$, then $L=\Z[c_{01},c_{10}, c_{11}...]/I$ then the universal formal power series becomes the universal formal group law $F_{Univ}$ over $L$.
\end{proof}


With the universal complex orientation, there is a graded formal group law $F_{MU}$ over $MU^*$ of degree $2$, or equivalently $MU_*$ of degree $-2$. Equip $\Z[c_{01},c_{10}, c_{11}...]$ with the grading $|c_{ij}|=2(i+j-1)$, which induces a grading on $L$, denotes $L_*$. There exists a map of graded ring 
\[
L_*\to \pi_*MU=MU_*
\]determined by the formal group law $F_{MU}$. Moreover, suppose we have a ring map $\phi:MU\to E$ determining a complex orientation on $E$, then there is identification $\phi^*F_{MU}=F_E$.

\begin{theorem}
    \cite{ChrQuillen}\cite{PreRavenelGreen}
    The graded ring map $L_*\to \pi_*MU$ that takes the universal formal group law $F_{Univ}$ to $F_{MU}$ is an isomorphism of graded ring. In particular, the ring $\pi_*MU$ classifies graded formal group laws of degree $-2$.
\end{theorem}
The earliest references for the ring structure of $\pi_*MU$ is given by Milnor \cite{ChrMilnorCobordism} and Novikov \cite{ChrNovikov2} \cite{ChrNovikov3}. However, we earliest identification of the ring structure through the universal formal group law perspective was given by Quillen \cite{ChrQuillen}.

\begin{definition}
    Given a formal group law $F$ over $R$, a \textbf{logarithm} of $F$ is an invertible formal group law $g\in R[[t]]$ such that
    \[
    F(x,y)=g^{-1}(g(x)+g(y))
    \]
\end{definition}
\begin{lemma}
    Let $R$ be an $\mathbb{Q}$-algebra. Suppose $g_1$ and $g_2$ are strict logarithm of $F$ over $R$, then 
    \[
    g_1=g_2
    \]
\end{lemma}
\begin{proof}
    Denote $h=g_1\circ g_2^{-1}$, then we have 
    \begin{align*}
        h(x+y)=h(x)+h(y)
    \end{align*}Differentiating respect to $y$ and let $y=0$, we have 
    \[
    h'(x)=h'(0)=1
    \]so $h(x)=x$ as it is an invertible power series.
\end{proof}
\begin{lemma}
   \cite{PreRavenelGreen}
    Let $R$ be an $\mathbb{Q}$-algebra, and let $F$ be a formal group law over $R$. Then there exists an isomorphism of formal group law $g: F\to G_{add}$, where $G_{add}(x,y)=x+y$ over $R$.
\end{lemma}
\begin{proof}
    Define logarithm of $F$,
    \[
    g(x):=\int_{0}^x \frac{dt}{F_1(t, 0)}
    \]where $F_1(x,y)=\partial F(x,y)/\partial y$. Notice 
    \[
    F_1(t,0)=1+a_{11}t+a_{21}t^2+a_{31}t^3+...
    \]and since the constant term is $1$, the multiplicative inverse exists. Moreover, $g(0)=0$, and $g'(0)=F_1(0,0)^{-1}=1$, so it is an invertible formal power series. Let $w=g(F(x,y))-g(x)-g(y)$, we want to show that this is zero.

    By associativity, $F(F(x,y),z)=F(x,F(y,z))$, we differentiate respect to $z$, we get 
    \[
    F_1(F(x,y), 0)=F_1(x, y) F_1(y,0)
    \]On the other hand,
    \[
    \frac{\partial w}{\partial y}= g'(F(x,y))F_1(x,y)-g'(y)
    \\=\frac{F_1(x,y)}{F_1(F(x,y), 0)}-\frac{1}{F_1(y, 0)}
    \\=\frac{F_1(x,y)F_1(y, 0)-F_1(F(x,y), 0)}{F_1(F(x,y), 0)F_1(y, 0)}
    \\=0
    \]so $w$ is constant. Take $x=y=0$, $w=0$.
\end{proof}

The algebra $L_*\otimes \mathbb{Q}$ classifies the graded formal group laws over $\mathbb{Q}$-algebra
\begin{lemma}
    \cite{PreRavenelGreen}
    Let $R_*=\mathbb{Q}[m_1,m_2,...]$ for $|m_i|=2i$ and let $f(x)=x+\sum _i m_ix^{i+1}$. There is $L_*\otimes \mathbb{Q}=R_*$ with $F_{Univ}\otimes \mathbb{Q}=f^{-1}(f(x)+f(y))$.
\end{lemma}
\begin{proof}
    For every formal group law $F$ over an $\mathbb{Q}$-algebra $T$, there is the logarithm $g$, which is classifies by an unique map
    \[
    \phi: R_*\to T
    \]such that $\phi^*f(x)=g(x)$. In particular, let $T=L_*\otimes \mathbb{Q}$ with the logarithm $g_{Univ}$ classified by $\phi_{Univ}: R_*\to L_*\otimes\mathbb{Q}$. 

    On the other hand, there is a formal group law over $R_*$ $f^{-1}(f(x)+f(y))$, so there is a map 
    \[
  \theta:  L_*\otimes \mathbb{Q}\to R_*
    \]
    Since $\phi^*_{Univ}f(x)=g_{Univ}(x)$, then $\phi^*_{Univ}f^{-1}(f(x)+f(y))=g_{Univ}^{-1}(g_{Univ}(x)+g_{Univ}(y))=F_{Univ}$. Therefore, $\theta$ and $\phi$ are inverses of each other.

\end{proof}

\begin{proof}
    The surjectivity of the map $\theta: L_*\to \pi_*MU$ would follow if we are able to show that the map $I/I^2\to J/J^2$ is surjective, where $I\subset L_*$ and $J\subset \pi_*MU$ are the ideal of elements in positive degree. Moreover, everything are finitely generated over $\Z$, we just need to show the surjectivity on the $p$ completion for all prime $p$.

    We have $(I/I^2)_{2i}\cong \Z_p\{a_i\}$, $(J/J^2)_{2n}=\Z_p\{t_i\}$, and $(K/K^2)_{2i}= \Z_p\{b_i\}$. The image of the map $\theta(a_i)= \lambda t_i +O(t_\alpha)$ for $O(t_\alpha)\in J^2$. Therefore, we just need to show $\lambda$ is invertible in $\Z_p$.  

    We know mod $K^2$
    \begin{align*}
     h\theta(a_i)=
            \begin{cases}
            pb_i, \text{ if $i=p^k-1$ for some prime $p$}
            \\ b_i, \text{ otherwise}
        \end{cases}
       \\ =  \lambda h(t_i)
       \\=  \begin{cases}
            \lambda pb_i, \text{ if $i=p^k-1$ for some prime $p$}
            \\ \lambda b_i, \text{ otherwise}
        \end{cases}
    \end{align*}And we are done.
\end{proof}
