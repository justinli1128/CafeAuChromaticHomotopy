\section{The Chromatic Tower and Chromatic Convergence}\label{sec:chromatic-results-chromaticconvergence}
% Write your notes here.
\begin{theorem}
    Let $X$ be a finite spectrum, then the chromatic tower
    \[
    L_{(p)}X\to ... \to L_nX\to L_{n-1}X\to ...\to L_1 X\to L_0X\simeq H\mathbb{Q}\wedge X
    \]converges
\end{theorem}
