\section{Formal Lie Variety and Formal Group}\label{sec:formalgroups-formalLie}

\begin{definition}
    Given a morphism of scheme $f: X\to Y$ and a quasicoherent $\Ocal_X$-module $\F$, a $f^{-1}\Ocal_Y$-derivation from $\Ocal_X$ to $\F$ is a $f^{-1}\Ocal_Y$-module morphism $d$ such that
    \[
    d(fg)=fdg+gdf
    \]for local section $f, g\in \Ocal_X(U)$ over $U\subseteq X$.

    Given a morphism of scheme $f: X\to Y$, the \textbf{Kahler differential} is the quasicoherent $\Ocal_X$-module $\Omega_{X/Y}$ such that for any $\Ocal_X$-module
    \[
    \Hom_{\Ocal_X}(\Omega_{X/Y}, \F)\cong \mathrm{Der}_{f^{-1}\Ocal_Y}(\Ocal_X, \F)
    \]in other words, $\Omega_{X/Y}$ classifies $f^{-1}\Ocal_Y$-derivations.
\end{definition}

For $S$-scheme closed immersion $i:X\to Y$ with ideal $\mathcal{I}_X$, there is an exact sequence 
\begin{align*}
   \omega_i=i^*\mathcal{I}_X\mathcal{I}_X^2\to i^*\Omega_{Y/S}\to \Omega_{X/S}\to 0
\end{align*}that comes from comparing differentials.

Suppose we have a closed immersion $i: X\to Y$ with defining ideal $\mathcal{I}_X$, there is a chain of closed subschemes
\[
Y_0=\mathrm{Inf}_X^0(Y)=X\subseteq Y_1=\mathrm{Inf}_X^1(Y)\subseteq Y_2=\mathrm{Inf}_X^2(Y)\subseteq ... \subseteq Y
\]
For each inclusion $i_n: X\to Y_n$ $n>0$, the conormal sheaf $\omega_{i_n}=i_n^*(\mathcal{I}_{i_n}/\mathcal{I}_{i_n}^2)\cong i_n^*(\mathcal{I}_{X}/\mathcal{I}_{X}^2)=\omega_i$.
Therefore, knowing $\omega_{i_1}$ tells us everything. In order for us to have information on the chain of closed subschemes, we define graded $\Ocal_{Y}/\mathcal{I}$-algebra $\mathrm{gr}_*(Y)$, such that
\begin{align*}
    \mathrm{gr}_k(Y):=i^*(\mathcal{I}_X^k/\mathcal{I}_X^{k+1})
\end{align*}
and the algebra structure comes from
\begin{align*}
 \mathrm{gr}_k(Y)\otimes_{\mathcal{O}_Y/\mathcal{I}}    \mathrm{gr}_{k'}(X)=i^*(\mathcal{I}_X^k/\mathcal{I}_X^{k+1})\otimes _{\Ocal_Y/\mathcal{I}_X}i^*(\mathcal{I}_X^{k'}/\mathcal{I}_X^{k'+1})
    \\\to i^*(\mathcal{I}_X^k/\mathcal{I}_X^{k+1}\otimes _{\Ocal_X}\mathcal{I}_X^{k'}/\mathcal{I}_X^{k'+1})
    \\ \to i^*(\mathcal{I}_X^{k+k'}/\mathcal{I}_X^{k+k'+1})
\end{align*}

Notice, $\mathrm{gr}_1(Y)= \omega_i$.

Suppose we have a closed immersion $i: S\to X$ over $S$ itself, with defining ideal $\mathcal{I}_S$, then the morphism 
\begin{align*}
    \omega_i\to i^*\Omega_{X/S}
\end{align*}becomes an isomorphism.

\begin{definition}
    A \textbf{formal Lie variety} is fpqc-sheaf $X$ over $S$, with section $e: S\to X$, such that 
    \begin{itemize}
        \item $X$ is ind-infinitesimal, with $\mathrm{Inf}_S^n(X)$ are representables are affines over $S$.
        \item $\omega_e$ is locally free of finite rank over $S$.
        \item The canonical map $\mathrm{Sym}_*\omega_e\to \mathrm{gr}_*(X)$ is an isomorphism.
    \end{itemize}
    A morphism of formal Lie variety is a morphism of fpqc-sheaves that preserves the section.
\end{definition}

\begin{definition}
    Suppose we have a formal Lie variety $(X,e)$ over $S$, and we have $f:T\to S$ is a scheme morphism. The pullback formal Lie variety is $(f^*X, f^*e)$.
\end{definition}

\begin{remark}
    Let $S=\Spec(A)$, the section $(0):\Spec(A)\to X=\Spf(A[x_1,...,x_n])$ of inclusion of the origin determines a formal Lie variety. We check each criteria one by one.

    First, $\mathrm{Inf}_S^n(X)=\Spec(A[x_1,...,x_n]/(x_1,...,x_n)^{n+1})$ and $\colim_n \mathrm{Inf}_S^n(X)=X$. Check.

    The module $(x_1,...,x_n)/(x_1,...,x_n)^{2}$ has generator $x_i$, so $\omega$ is a $A$-module of rank $n$.

    The multiplication of $\mathrm{gr}_k(X)$ is precisely the multiplication of $x_i$'s, so the third isomorphism is checked.
\end{remark}
\begin{proposition}
    Let $(X,e)$ be a formal Lie variety over $S=\Spec(A)$. There exists an isomorphism $(\Spf(A[x_1,...,x_n]), (0))\cong (X,e)$ where $n=\mathrm{rk}\omega_e$.
\end{proposition}

\begin{proof}
    Since $X_n$ are affines, they are completely described by their global sections $R_n$, i.e. $\Spec(R_n)\cong X_n$.

    We know there is exact sequence
    \begin{align*}
        0\to I_n\to R_n\to A\to 0
    \end{align*}

    The map $R_n\to R_{n-1}$ is surjective, as it is modulo some ideal in $R_n$, which implies $I_n\to I_{n-1}$ is surjective. 

    The module $I_1$ is the kernel of $R_1\to A$, which implies $I_1\cong \omega_e$. There exists splitting $\omega_e\to I_n\to \omega_e$ due to the surjection, which induces an algebra morphism
    \[
    \mathrm{Sym}\omega_e\to R_n
    \]
    Choose generator $x_i$ in $\omega_e$, the freeness implies $\omega_e\cong A\{x_i\}$ and $\mathrm{Sym}(\omega_e)\cong A[x_1,...,x_n]$. Notices, $I_n^{n+1}=0$ in $R_n$, so this factors through
    \[
     A[x_1,...,x_n]/I_n^{n+1}\to R_n
    \]
    
    The augmentation $R_n\to A$ has kernel $I_n$, which implies that the map above is surjective, because every elements in $R_n$ splits into $r+a$ for $r\in R_n- I_n$ and $a\in A$. Moreover, this is an isomorphism, because $\mathrm{Sym}_k \omega_e\cong \mathcal{I}_n^{k}/\mathcal{I}_n^{k+1}$.

    The ideal $I$ is $(x_1,...,x_n)$ by looking at the $n=1$ case, so we have 
    \[
    A[x_1,...,x_n]/(x_1,...,x_n)^{n+1}\to R_n
    \]is an isomorphism.
\end{proof}

Moreover, the affine assumption implies that for every general scheme $S$, the formal Lie variety over $S$ are glued together by formal infinitesimal neighbourhood.

However, the isomorphism of the identification is very noncanonical, so there is some ambiguity in the theory.

\begin{definition}
    Given a scheme $S$, a \textbf{formal group} over $S$ is a group object in the category of formal Lie varieties such that $\omega_e$ is locally free of rank $1$, and a \textbf{morphism of formal group} is a formal Lie variety morphism that preserves the group structure.
\end{definition}

\begin{remark}
    Recall our identification of formal Lie varieties over affines implies a formal group over $\Spec(A)$ can be identified with a group structure on $\Spec(A[t])$, or equivalently a formal group law over $A$.
\end{remark}
