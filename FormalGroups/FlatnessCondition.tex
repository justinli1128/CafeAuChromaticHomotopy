\section{A Flatness Condition on Tensor Product of Quasicoherent Modules}\label{sec:formalgroups-flatness}

Fix a prime $p$, let $v_i\in L$ where $v_0=p$ to be the coefficient of the $p$-series of universal formal group over $L$.

\begin{theorem}
    \cite{PreLandweber}
    Let $M$ be a $L$-module, or equivalently a quasicoherent $\Spec(L)$-module $\tilde {M}$, then the functor
    \begin{align*}
        \Qcoh_{\Mfg}\xrightarrow{q^*}\Qcoh_{\Spec(L)}\xrightarrow{ -\otimes \tilde {M}}\Qcoh_{\Spec(L)}
    \end{align*}is exact if the following condition is satisfied: there exists a prime $p$, such that the sequence of element $(p,v_1,v_2,...)$ is a regular sequence of $M$.
\end{theorem}

\begin{proof}
    Since $\Mfg\simeq [\Spec(L)/\Spec(LB[u^{\pm}])]$, the statement is equivalent to showing that for any exact sequence of $(L, LB[u^{\pm}])$-comodule 
    \[
    0\to N' \to N\to N''\to 0
    \]the sequence
    \[
    0\to N'\otimes_L M\to N\otimes_L M\to  N''\otimes_L M\to 0
    \]is exact.

    This is equivalent to saying $\mathrm{Tor}^i_L(N, M)=0$ for any comodule $N$.
