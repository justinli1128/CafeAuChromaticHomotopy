\begin{definition}
    Let $X$ be a topological space, the \textbf{category of presheaves} over $X$ is denoted as $\pSh(X)$, and the \textbf{category of sheaves} over $X$ is denoted as $\Sh(X)$. 
    Let $C$ be a category that is bicomplete, the \textbf{category of presheaves valued in $C$} is denoted as $\pSh(X;C)$, and the \textbf{category of sheaves valued in $C$} is denoted as $\Sch(X;C)$.

    Let $f: X\to Y$ be a continuous map.
    
    Denote the \textbf{direct image functor} $f_*: \Sh(X;C)\to \Sh(Y;C)$ on $\F\in \Sh(X;C)$
    \begin{align*}
        f^*\F: O(Y)^{op}\to C 
        \\ V\mapsto \F(f^{-1}(V))
    \end{align*}with the morphisms between presheaves over $Y$ defined at $V$ to be the morphism between presheaves over $X$ at $f^{-1}(V)$.
\end{definition}


\begin{definition}
A \textbf{locally ringed space} is a space $X$ with a sheaf of rings
$\Ocal_X$ whose stalks $\Ocal_{X,x}$ are local rings with maximal ideal $ \mathfrak{m}_{X,x}$. A \textbf{locally ringed space morphisms} is a continuous map
$f:X\to Y$ with a sheaf map $f^{\shp}: \Ocal_Y\to f_*\Ocal_X$ inducing local
homomorphisms on stalks, that is $f^{\shp}(\mathfrak{m}_{Y,f(x)}\subseteq \mathfrak{m}_{X,x}$.
\end{definition}

\begin{definition}
    Let $A$ be a commutative ring, the \textbf{spectrum} $\Spec(A)$ is the set of strict prime ideals of $A$. The \textbf{standard closed sets} of $\Spec(A)$ are the sets $V(f)$ for $f\in A$ such that
    \[
    V(f):=\{\mathfrak{p}\in \Spec(A): f\in \mathfrak{p}\}
    \]and the \textbf{standard open sets} are $D(f)=V(f)^c$, that is 
    \[
    D(f):=\{\mathfrak{p}\in \Spec(A): f\notin \mathfrak{p}\}
    \]The \tetxbf{Zariski topology} of $\Spec(A)$ is the topology generated by $D(f)$ for all $f \in A$.

    Let $\varphi: B\to A$ be a ring morphism, define continuous map $f:\Spec(A)\to \Spec(B)$ as $f(\mathfrak{p})=\varphi^{-1}(\mathfrak{p})$.
\end{definition}
We see that $D(0)=\emptyset$ and $D(1)=\Spec(A)$.

Let $I\subseteq A$ be an ideal of $A$, then we define $V(I):=\{\mathfrak{p}: I\subset \mathfrak{p}\}$ and $D(I)=V(I)^c=\{\mathfrak{p}: I\not\subset \mathfrak{p}\}$.
 Notice, for $I=\langle f_1,f_2,...,f_n\rangle$, $V(I)= V(f_1)\cap V(f_2)\cap...\cap V(f_n)$ and $D(I)=D(f_1)\cup D(f_2)\cup...\cup D(f_n)$. Moreover, $V(f)\cup V(g)=V(fg)$ by the characterization of prime ideals, and $D(f)\cap D(g)=D(fg)$.


\begin{definition}
    Let $A$ be a commutative ring, and let $S\subseteq A$ be a subset of $A$ that is closed under multiplication, we define an equivalence relation on $A\times S$ via 
    \[
    (a,s)\sim (a',s')\text{ if } as'=a's
    \]
    Let $A_S$ be the equivalent class $A\times S/\sim$, and we define addition and multiplication on $A_S$ as
    \begin{align*}
        [(a,s)]\cdot[(b,t)]=[(ab, st)]
        \\ [(a,s)]\cdot[(b,t)]=[(at+bs, st)]
    \end{align*}which determines a commutative ring. For $f\in A$, we define $A_f=A_S$, where $S=\{f^n: n\in \mathbb{N}\}=\{1,f,f^2,...\} $.
\end{definition}

\begin{definition}
    The \textbf{structure sheaf} of $\Spec(A)$, denoted as $\Ocal_{\Spec(A)}$ to be the sheaf of rings over $\Spec(A)$, such that on $D(f)$
    \begin{align*}
    \Ocal_{\Spec(A)}(D(f)):=A_{f}
    \end{align*}and for the restriction $\Ocal_{\Spec(A)}(D(f))\to    \Ocal_{\Spec(A)}(D(fg))$, we define $[(a,f^n)]\mapsto [(ag^n, f^ng^n)]$.
\end{definition}

\begin{example} 
    
    If $k$ is an algebraically closed field, then the prime ideals of $\Spec(k[x_1,x_2,...,x_n])$ are either $0$ or $\langle f_1,f_2,...,f_m\rangle$ for some irreducible polynomials $f_i$, and the maximal ideals are $\langle x_1-a_1,x_2-a_2,...,x_n-a_n\rangle$ for $a_i \in k$.
    
    We define function
    \begin{align*}
        \A^n_k\to \Spec(k[x_1,x_2,...,x_n])-\{0\}
        \\ (a_1,a_2,...,a_n)\mapsto \langle x_1-a_1,x_2-a_2,...,x_n-a_n\rangle
    \end{align*}
    By Hilbert's Nullstellensatz, this is an homeomorphism of topological space.

    In general, let $k$ be a commutative ring, $\A_k^n:=\Spec(k[x_1,x_2,...,x_n])$. If $k$ is not algebraically closed, there is not a homeomorphism above, for instance let $k=\mathbb{R}$, $n=1$ and $\mathfrak{p}=\langle x^2+1\rangle$.
\end{example}

\begin{definition}
    An \textbf{affine scheme} is a locally ringed space isomorphic to $(\Spec A,\Ocal_{\Spec A})$. A \textbf{scheme} is a locally ringed space with an open cover $U_i$ such that $(U_i,\Ocal_X|_{U_i})$ are isomorphic to some affine scheme $\Spec(A_i)$.
Write $\Sch$ for the category of schemes and $\Aff$ for the category of affine schemes.
\end{definition}

We denote $\kappa(x):=\Ocal_{X,x}/\mathfrak{m}_x$ the field of \textbf{rational functions} at $x$.

\begin{theorem}
    \cite{AGHartshorne}\cite{AGVakil}
There exists adjunction
\begin{align*}
    \Gamma\vdash\Spec: \Sch\to \Ring^{op}
\end{align*}So there is
\[
    \Hom_{\Ring}(A,\Gamma(X,\Ocal_X) )\cong \Hom_{\Sch}(X, \Spec (A))
\]
In particular, this induces equivalence $\Aff\to \Ring^{op}$.
\end{theorem}

\begin{definition}\cite{AGGoerss}
The \textbf{functor of points} of $X$ is
$h_X:\Aff^{\mathrm{op}}=\Ring\to \Sets$, $h_X(A)=\Hom_{\Sch}(\Spec(A),X)$.
\end{definition}

\begin{example}
For $X=\A^n_{\Z}$, the functor of point $h_X(A)=\Hom_{\Sch}(\Spec(A),\A^n_{\Z})\cong \Hom_{\Sch}(\Z[x_1,...,x_n], A)$. Therefore, the functor of points of $\A^n_{\Z}$ classifies $n$-elements of $A$.

Define $\mathbb{G}_m:=\A^1_\Z-\{0\}=\Spec(\Z[x,x^{-1}])$, the $h_{\mathbb{G}_m}(A)=\Hom_{\Sch}(\Z[x,x^{-1}], A)=A^\times$.
\end{example}

There exists a Grothendieck topology on $\Sch^{op}$ we call the Zariski topology, such that every scheme 
admits an cover of objects in $\Aff^{op}$. Moreover, the functor of points of a scheme is a sheaf respect to the Zariski topology on $\Aff^{op}$, 
hence speaking of a functor of point of a scheme, we may talk about either over $\Aff^{op}$ or over $\Sch^{op}$ and they are equivalent.

\begin{definition}
A scheme is \textbf{quasicompact, connected, or irreducible} if its underlying space is quasicompact, connected, or irreducible. 
The \textbf{dimension} of a scheme is the dimension of the underlying space.
\end{definition}

\begin{definition}
A scheme is \textbf{reduced} if its stalk, or equivalently over all opens, have no nonzero nilpotent. Moreover, a scheme is \textbf{integral} if it is reduced and irreducible.

Let $\Spec(A_i)$ be an affine open cover of an integral scheme $X$, therefore $A_i$ are integral domains, and each $\Spec(A_i)$, and hence $X$, have the unique generic point $\eta$.

Define $K(X):=\mathrm{Frac}(A_i)$ to be the quotient ring of one of the $A_i$, and we refer to it as the \textbf{function field}. Notice $K(X)$ is independent of $A_i$.
\end{definition}
