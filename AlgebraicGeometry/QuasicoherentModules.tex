\section{Quasicoherent modules}\label{sec:algebraicgeometry-quasicoherentmodules}

\begin{definition}
    Let $(C, \Ocal_C)$ be a ringed site, a $\Ocal_C$-module is a sheaf of abelian group $\F$ with map $\varphi:\Ocal_C \otimes \F \to \F$ such that given  $U \in C$,
    \[
   \varphi_U: \Ocal_C(U)\otimes \F(U)\to \F(U)
    \]determines a $\Ocal_C(U)$-module, and given $V\to U$ 
    \begin{center}
\begin{tikzcd}
 \Ocal_C(U)\otimes \F(U) \arrow[d, "\varphi_U"] \arrow[r] &  \Ocal_C(V)\otimes \F(V) \arrow[d, "\varphi_V"] \\
\F(U) \arrow[r]                                           & \F(V)                                          
\end{tikzcd}
    \end{center}commutes. Denote $\Module_C$ the category of $\Ocal_C$-modules. 
\end{definition}

\begin{definition}
    Let $A$ be a ring, and $M$ be an $A$-module, denote $\tilde{M}$ the $\Ocal_{\Spec(A)}$-module such that 
    \[
    \tilde{M}(D(f))=M_f=M\otimes A_f
    \]with the induced $\Ocal_{\Spec(A)}(D(f))=A_f$-module structure.
\end{definition}

\begin{definition}
    Let $X$ be a scheme, a $\mathcal{O}_X$-module $\F$ is called a \textbf{quasicoherent module over $X$} if there is an affine open cover $\Spec(A_i)$, and $A_i$-modules $M_i$, such that 
    \[
    \F|_{\Spec(A_i)}\cong \tilde{M_i}
    \]
\end{definition}

We denote the category of quasicoherent modules over $X$ as $\Qcoh_X$. 

We make a similar definition that would make the generalization over stack easier.

\begin{definition}
    Let $(C, \Ocal_C)$ be a ringed site, a \textbf{globally presented module} over $C$ is a $\Ocal_C$-module $\F$ such that there is exact sequence
    \[
    \Ocal_C^{\oplus i}\to \Ocal_{C}^{\oplus j}\to \F\to 0
    \]and a $\Ocal_C$-module is \textbf{quasicoherent} if for every $U\in C$ there is a covering $U_i\to U$ such that $\F|_{C_{/U_i}}$ are globally presented $\Ocal_{C_{/U_i}}$-module. 
\end{definition}

Let $(\M, \Ocal_{\M})$ be the ringed site determined by the fpqc-stack over a scheme $S$, we can make a similar construction.
\begin{definition}
    Let $(\M, \Ocal_{\M})$ be the ringes site determined by the fpqc-stack over a scheme $S$, a \textbf{cartesian module} over $\M$ is a $\Ocal_{\M}$-module $\F$
    such that for $f: x\to y$ in $\M$ living over $\Spec(B)\to \Spec(A)$, there is isomorphism
    \[
        \F(y)\otimes_A B \to \F(x)
    \]Denote $\Module^{cart}_{\M}$ the category of cartesian module over $\M$.
\end{definition}

Notice, just requiring the tensor product property would already implies the fpqc-sheaf property. Take a faithfully flat algebra $A\to B$, the sheaf condition is then equivalent to exactness of 
\[
0 \to M(\Spec(A)) \to B\otimes_A M(\Spec(A)) \to B\otimes_AB\otimes _A M(\Spec(A))
\]which is implied.

\begin{proposition}
Let $(\M, \Ocal_{\M})$ be the ringes site determined by the fpqc-stack over a scheme $S$, a $\Ocal_{\M}$-module $\F$ is quasicoherent if and only if it is cartesian.

\end{proposition}
\begin{proof}
    First suppose it is cartesian, for every $x\in \M$ living over $\Spec(A)$, there is a $A$-module exact sequence
    \[
    A^{\oplus I}\to A^{\oplus J}\to M(x)\to 0
    \]
    For $y\to x$ living over $\Spec(B)\to \Spec(A)$, there is then exact sequence
    \[
    B^{\oplus I}\to B^{\oplus J}\to B\otimes_AM(x)\cong M(y)\to 0
    \]
    which implies the quasicoherentness.

    On the other hand, suppose we have quasicoherent module $\F$, let $x \in \M$ living over $\Spec(A)$ and let $\{x_i\to x\}$ over $\{\Spec(A_i)\to \Spec(A)\}$ be an cover, for which $\Spec(A_i)$ determines a cover of $\Spec(A)$, and exact sequences
    \[
     \Ocal_{x_i}^{\oplus I_i}\to \Ocal_{x_i}^{\oplus J_i}\to \F|_{x_i}}\to 0
    \]
    Suppose we have $y\to x$ over $\Spec(B)\to \Spec(A)$, then $\{y_i:=y\times_{x}x_i\to y\}$ over $B_i:=B\otimes_A A_i$ is a cover of $y$, and there is exact sequence
    \[
     \Ocal_{y_i}^{\oplus I_i}\to \Ocal_{y_i}^{\oplus J_i}\to \F|_{y_i}}\to 0
    \]determined by restriction on $y_i$.

    Since $\Ocal_{x_i}$ are cartesian, there exists isomorphism of exact sequences
   \begin{center}

    \begin{tikzcd}
B_i\otimes_{A_i}\Ocal_{x_i}^{\oplus I_i}(x_i) \arrow[d, "\cong"] \arrow[r] & B_i\otimes_{A_i}\Ocal_{x_i}^{\oplus J_i}(x_i) \arrow[d, "\cong"] \arrow[r] & B_i\otimes_{A_i} \F(x_i) \arrow[r] \arrow[d, dotted] & 0 \\
B_i^{\oplus I_i} \arrow[r]                                                 & B_i^{\oplus J_i} \arrow[r]                                                 & \F(y_i) \arrow[r]                                    & 0
\end{tikzcd}

\end{center}which implies isomorphism $ B_i\otimes_{A_i}\F(x_i)\to \F(y_i)$. Hence we have isomorphism $B\otimes_{A}\F(x)\to \F(y)$ from the sheaf property.
\end{proof}
